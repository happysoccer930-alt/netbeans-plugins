// Registra un depósito y actualiza el saldo del cliente.
@Service
public class DepositoService {

    private final CuentaRepository cuentas;
    private static final BigDecimal LIMITE = new BigDecimal("15000.00");

    public Deposito registrar(String dni, BigDecimal monto) {
        if (monto.compareTo(LIMITE) > 0) {
            throw new LimiteExcedido("Monto " + monto);   
        }
        Cuenta cuenta = cuentas.buscarPorDni(dni);
        return cuenta.acreditar(monto, "ventanilla");
    }
}
